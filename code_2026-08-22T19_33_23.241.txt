local function CreateInstance(Class, Properties)
    local Instance = Instance.new(Class)
    for Prop, Value in pairs(Properties) do
        if Prop ~= "Parent" then
            pcall(function()
                Instance[Prop] = Value
            end)
        end
    end
    if Properties.Parent then
        Instance.Parent = Properties.Parent
    end
    return Instance
end

    end)
end

-- ============ 核心 API 调用逻辑 (唯一修改处) ============
local function CallSenseNovaAPI(prompt, apiKey, callback)
    if not apiKey or apiKey == "" then
        callback(false, "请先输入 API Key")
        return
    end

    local url = API_CONFIG.BaseURL .. API_CONFIG.Endpoint
    local headers = {
        ["Content-Type"] = "application/json",
        ["Authorization"] = "Bearer " .. apiKey
    }

    -- 构建 System Prompt，强制输出纯 Lua 代码
    local systemPrompt = [[你是 Roblox Lua 脚本专家。请根据用户需求生成完整、可直接运行的 Luau 代码。
要求：
1. 只输出代码，不要任何 markdown 代码块标记，不要解释性文字。
2. 使用 game:GetService 获取服务，避免全局变量污染。
3. 处理 CharacterAdded 等生命周期事件。
4. 包含基本的错误处理 (pcall)。
5. 适配常见执行器环境。]]

    local body = {
        model = API_CONFIG.Model,
        messages = {
            {role = "system", content = systemPrompt},
            {role = "user", content = "类型: " .. SelectedType .. "\n需求: " .. prompt}
        },
        temperature = API_CONFIG.Temperature,
        max_tokens = API_CONFIG.MaxTokens,
        stream = false
    }

    local success, result = pcall(function()
        return HttpService:RequestAsync({
            Url = url,
            Method = "POST",
            Headers = headers,
            Body = HttpService:JSONEncode(body)
        })
    end)

    if not success then
        callback(false, "网络请求失败: " .. tostring(result))
        return
    end

    if result.StatusCode ~= 200 then
        callback(false, "API 错误 " .. result.StatusCode .. ": " .. result.Body)
        return
    end

    local decodeSuccess, data = pcall(function()
        return HttpService:JSONDecode(result.Body)
    end)

    if not decodeSuccess or not data.choices or not data.choices[1] or not data.choices[1].message then
        callback(false, "响应格式异常: " .. result.Body)
        return
    end

    local code = data.choices[1].message.content
    -- 清理可能残留的 markdown 标记
    code = code:gsub("^```[%w]*\n", ""):gsub("\n```$", ""):gsub("^```", ""):gsub("```$", "")
    
    callback(true, code)
end

GenerateBtn.MouseButton1Click:Connect(function()
    local Desc = DescInput.Text
    local ApiKey = ApiKeyInput.Text
    
    if Desc == "" then
        ShowNotification("⚠️ 警告", "请输入脚本需求描述")
        return
    end
    
    if ApiKey == "" then
        ShowNotification("⚠️ 警告", "请输入 API Key")
        return
    end

    GenerateBtn.Text = "⏳ 生成中..."
    GenerateBtn.BackgroundColor3 = Colors.Accent
    GenerateBtn.AutoButtonColor = false
    StatusLabel.Text = "正在调用 SenseNova API..."
    ResultFrame.Visible = false

    task.spawn(function()
        CallSenseNovaAPI(Desc, ApiKey, function(success, response)
            -- 回到主线程更新 UI
            task.defer(function()
                GenerateBtn.Text = "✨ 生成脚本 (调用 API)"
                GenerateBtn.BackgroundColor3 = Colors.Success
                
                if success then
                    StatusLabel.Text = "✅ 生成成功 | " .. os.date("%H:%M:%S")
                    ResultFrame.Visible = true
                    ResultText.Text = response
                    
                    -- 自动保存到历史
                    table.insert(_G.AIScript.SavedScripts, {
                        Name = (SelectedType .. ": " .. Desc):sub(1, 30),
                        Code = response,
                        Time = os.date("%H:%M:%S"),
                        Type = SelectedType
                    })
                    
                    ShowNotification("✅ 成功", "脚本已生成并保存到历史")
                else
                    StatusLabel.Text = "❌ 失败: " .. response:sub(1, 50)
                    ShowNotification("❌ 生成失败", response)
                end
            end)
        end)
    end)
end)

local function RefreshSavedList()
    for _, Child in ipairs(SavedListFrame:GetChildren()) do
        if Child:IsA("Frame") then Child:Destroy() end
    end
    
    if #_G.AIScript.SavedScripts == 0 then
        local EmptyLabel = CreateInstance("TextLabel", {
            Text = "暂无已保存的脚本\n点击「AI生成」标签页创建你的第一个脚本",
            TextSize = 14,
            TextColor3 = Colors.Text,
            BackgroundTransparency = 1,
            Size = UDim2.new(1, 0, 0, 60),
            Parent = SavedListFrame
        })
        SavedListFrame.CanvasSize = UDim2.new(0, 0, 0, 60)
        return
    end
    
    local totalHeight = 0
    for i, ScriptData in ipairs(_G.AIScript.SavedScripts) do
        local ScriptItem = CreateInstance("Frame", {
            BackgroundColor3 = Colors.Surface,
            Size = UDim2.new(1, 0, 0, 70),
            Parent = SavedListFrame,
            LayoutOrder = -i -- 最新在顶部
        })
        AddCorner(ScriptItem, 6)
        totalHeight = totalHeight + 78
        
        CreateInstance("TextLabel", {
            Text = "📝 " .. ScriptData.Name,
            TextSize = 13,
            TextColor3 = Colors.Accent,
            Font = Enum.Font.GothamBold,
            BackgroundTransparency = 1,
            Position = UDim2.new(0, 10, 0, 5),
            Size = UDim2.new(1, -120, 0, 20),
            TextXAlignment = Enum.TextXAlignment.Left,
            Parent = ScriptItem
        })
        
        CreateInstance("TextLabel", {
            Text = "⏰ " .. ScriptData.Time .. " | 类型: " .. (ScriptData.Type or "未知"),
            TextSize = 11,
            TextColor3 = Colors.Text,
            BackgroundTransparency = 1,
            Position = UDim2.new(0, 10, 0, 28),
            Size = UDim2.new(1, -120, 0, 18),
            TextXAlignment = Enum.TextXAlignment.Left,
            Parent = ScriptItem
        })
        
        CreateInstance("TextLabel", {
            Text = ScriptData.Code:sub(1, 80) .. "...",
            TextSize = 10,
            TextColor3 = Color3.fromRGB(150, 150, 180),
            BackgroundTransparency = 1,
            Position = UDim2.new(0, 10, 0, 48),
            Size = UDim2.new(1, -120, 0, 18),
            TextXAlignment = Enum.TextXAlignment.Left,
            TextTruncate = Enum.TextTruncate.AtEnd,
            Parent = ScriptItem
        })
        
        -- 按钮组
        local BtnContainer = CreateInstance("Frame", {
            BackgroundTransparency = 1,
            Position = UDim2.new(1, -110, 0, 10),
            Size = UDim2.new(0, 100, 0, 50),
            Parent = ScriptItem
        })
        
        local LoadBtn = CreateInstance("TextButton", {
            Text = "📋 加载",
            TextSize = 10,
            TextColor3 = Color3.fromRGB(255, 255, 255),
            Font = Enum.Font.GothamBold,
            BackgroundColor3 = Colors.Primary,
            Position = UDim2.new(0, 0, 0, 0),
            Size = UDim2.new(0, 48, 0, 22),
            Parent = BtnContainer,
            AutoButtonColor = false
        })
        AddCorner(LoadBtn, 4)
        LoadBtn.MouseButton1Click:Connect(function()
            ResultFrame.Visible = true
            ResultText.Text = ScriptData.Code
            CurrentTab = "AIGen"
            TabPages["AIGen"].Page.Visible = true
            TabPages["Saved"].Page.Visible = false
            StatusLabel.Text = "📂 已加载历史脚本: " .. ScriptData.Name
        end)
        
        local DelBtn = CreateInstance("TextButton", {
            Text = "🗑️",
            TextSize = 12,
            BackgroundColor3 = Colors.Error,
            Position = UDim2.new(0, 52, 0, 0),
            Size = UDim2.new(0, 22, 0, 22),
            Parent = BtnContainer,
            AutoButtonColor = false
        })
        AddCorner(DelBtn, 4)
        DelBtn.MouseButton1Click:Connect(function()
            table.remove(_G.AIScript.SavedScripts, i)
            RefreshSavedList()
        end)
        
        local ExecBtn = CreateInstance("TextButton", {
            Text = "▶️",
            TextSize = 12,
            BackgroundColor3 = Colors.Success,
            Position = UDim2.new(0, 76, 0, 0),
            Size = UDim2.new(0, 22, 0, 22),
            Parent = BtnContainer,
            AutoButtonColor = false
        })
        AddCorner(ExecBtn, 4)
        ExecBtn.MouseButton1Click:Connect(function()
            pcall(function() loadstring(ScriptData.Code)() end)
            ShowNotification("▶️ 执行", "已尝试运行: " .. ScriptData.Name)
        end)
    end
    
    SavedListFrame.CanvasSize = UDim2.new(0, 0, 0, totalHeight)
end

RefreshSavedList()
