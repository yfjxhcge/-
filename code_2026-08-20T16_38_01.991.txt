-- HTTP 请求函数
local function HttpRequest(url, method, headers, body)
    local requestFunc = syn and syn.request or http_request or request
    if not requestFunc then
        return nil, "当前注入器不支持 HTTP 请求"
    end
    local success, response = pcall(requestFunc, {
        Url = url,
        Method = method or "GET",
        Headers = headers or {},
        Body = body and HttpService:JSONEncode(body) or nil
    })
    if success then
        return response
    else
        return nil, tostring(response)
    end
end

-- 发送消息到 AI
local function AskAI(prompt, callback)
    local headers = {
        ["Content-Type"] = "application/json",
        ["Authorization"] = "Bearer " .. CONFIG.API_KEY
    }
    local body = {
        model = CONFIG.MODEL,
        messages = {
            { role = "system", content = CONFIG.SYSTEM_PROMPT },
            { role = "user", content = prompt }
        },
        temperature = 0.3,
        max_tokens = 2048
    }
    
    local response, err = HttpRequest(CONFIG.API_URL, "POST", headers, body)
    if not response then
        callback(nil, "API 请求失败: " .. (err or "未知错误"))
        return
    end
    
    local success, data = pcall(HttpService.JSONDecode, HttpService, response.Body)
    if not success or not data then
        callback(nil, "解析响应失败: " .. tostring(response.Body))
        return
    end
    
    if data.choices and data.choices[1] then
        local content = data.choices[1].message.content
        -- 提取代码块（如果有）
        local code = content:match("```lua\n(.-)```") or content:match("```\n(.-)```") or content
        callback(code, nil)
    else
        callback(nil, "AI 返回异常: " .. HttpService:JSONEncode(data))
    end
end

-- 执行 Lua 代码（安全模式）
local function ExecuteScript(code)
    local func, err = loadstring(code)
    if not func then
        return false, "编译错误: " .. tostring(err)
    end
    local success, result = pcall(func)
    if not success then
        return false, "运行时错误: " .. tostring(result)
    end
    return true, result
end

local function CreateDrawObject(class, props)
    local obj = Drawing.new(class)
    for k, v in pairs(props) do
        obj[k] = v
    end
    return obj
end

local function IsMouseOver(obj)
    local mx, my = UIS:GetMouseLocation().X, UIS:GetMouseLocation().Y
    if obj.ClassName == "Square" or obj.ClassName == "Rectangle" then
        return mx >= obj.X and mx <= obj.X + obj.Width and my >= obj.Y and my <= obj.Y + obj.Height
    elseif obj.ClassName == "Circle" then
        local dx, dy = mx - obj.X, my - obj.Y
        return (dx*dx + dy*dy) <= obj.Radius * obj.Radius
    end
    return false
end

    if processed then return end

                if type(obj) == "table" and obj.Visible ~= nil then
                    obj.Visible = false
                end
            end
            return
        end
        
        -- 检查发送按钮
        if IsMouseOver(UI.SendBtn) and UserInput ~= "" then
            UI.StatusText.Text = "🤔 正在向 AI 发送请求..."

                if code then

                    UI.CodeText.Text = "错误: " .. (err or "未知错误")
                    UI.StatusText.Text = "❌ " .. (err or "生成失败")
                end
            end)
            return
        end
        
        -- 检查执行按钮
        if IsMouseOver(UI.ExecuteBtn) and UI.CodeText.Text ~= "等待 AI 生成代码..." and UI.CodeText.Text ~= "" then
            UI.StatusText.Text = "⚡ 正在执行脚本..."
            local success, result = ExecuteScript(UI.CodeText.Text)

                UI.CodeText.Text = UI.CodeText.Text .. "\n\n-- 错误: " .. tostring(result)
            end
            return
        end
        
        -- 检查清空按钮
        if IsMouseOver(UI.ClearBtn) then
            UI.CodeText.Text = "等待 AI 生成代码..."
            UI.StatusText.Text = "已清空 | 输入需求后点击生成"
            return
        end
        
        -- 检查输入框点击
        if IsMouseOver(UI.InputBg) then
            IsTyping = true
            UI.InputPlaceholder.Visible = false
            UI.StatusText.Text = "⌨️ 输入中... 按 Enter 发送"
            return

-- 键盘输入
UIS.InputBegan:Connect(function(input, processed)
    if processed then return end
    if not IsTyping then return end
    
    if input.UserInputType == Enum.UserInputType.Keyboard then
        local key = input.KeyCode
        
        if key == Enum.KeyCode.Backspace then
            UserInput = UserInput:sub(1, -2)
        elseif key == Enum.KeyCode.Return then
            IsTyping = false
            if UserInput ~= "" then
                UI.StatusText.Text = "🤔 正在向 AI 发送请求..."
                AskAI(UserInput, function(code, err)
                    if code then
                        UI.CodeText.Text = code
                        UI.StatusText.Text = "✅ 代码生成成功！点击「执行脚本」运行"
                    else
                        UI.CodeText.Text = "错误: " .. (err or "未知错误")
                        UI.StatusText.Text = "❌ " .. (err or "生成失败")
                    end
                end)
            end
        elseif key == Enum.KeyCode.Escape then
            IsTyping = false
            if UserInput == "" then
                UI.InputPlaceholder.Visible = true
            end
        else
            local char = input.KeyCode.Name
            -- 简单处理字符输入
            if #char == 1 then
                UserInput = UserInput .. char
            elseif char == "Space" then
                UserInput = UserInput .. " "
            end
        end
        
        UI.InputText.Text = UserInput
    end
end)
