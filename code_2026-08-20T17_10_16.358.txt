-- HTTP 请求（通用方式）
local function HttpRequest(url, method, headers, body)
    local requestFunc = http_request or request or (syn and syn.request)
    if not requestFunc then
        return nil, "不支持 HTTP 请求"
    end
    local success, response = pcall(requestFunc, {
        Url = url,
        Method = method or "GET",
        Headers = headers or {},
        Body = body and HttpService:JSONEncode(body) or nil
    })
    if success then
        return response
    end
    return nil, tostring(response)
end

-- 调用 AI
local function AskAI(prompt, callback)
    local headers = {
        ["Content-Type"] = "application/json",
        ["Authorization"] = "Bearer " .. CONFIG.API_KEY
    }
    local body = {
        model = CONFIG.MODEL,
        messages = {
            { role = "system", content = CONFIG.SYSTEM_PROMPT },
            { role = "user", content = prompt }
        },
        temperature = 0.3,
        max_tokens = 2048
    }
    
    local response, err = HttpRequest(CONFIG.API_URL, "POST", headers, body)
    if not response then
        callback(nil, "API请求失败: " .. (err or "未知错误"))
        return
    end
    
    local ok, data = pcall(HttpService.JSONDecode, HttpService, response.Body)
    if not ok or not data then
        callback(nil, "解析失败: " .. tostring(response.Body))
        return
    end
    
    if data.choices and data.choices[1] then
        local content = data.choices[1].message.content
        local code = content:match("```lua\n(.-)```") 
            or content:match("```\n(.-)```") 
            or content
        callback(code, nil)
    else
        callback(nil, "AI返回异常: " .. HttpService:JSONEncode(data))
    end
end

-- 执行代码
local function RunCode(code)
    local func, err = loadstring(code)
    if not func then
        return false, "编译错误: " .. tostring(err)
    end
    local ok, result = pcall(func)
    if not ok then
        return false, "运行时错误: " .. tostring(result)
    end
    return true, result
end

-- 创建 Drawing 对象
local function NewDraw(class, props)
    local obj = Drawing.new(class)
    if props then
        for k, v in pairs(props) do
            obj[k] = v
        end
    end
    return obj
end

-- 鼠标是否在对象上
local function IsOver(obj)
    local mx, my = UIS:GetMouseLocation().X, UIS:GetMouseLocation().Y
    if obj.ClassName == "Square" then
        return mx >= obj.X and mx <= obj.X + obj.Width 
           and my >= obj.Y and my <= obj.Y + obj.Height
    elseif obj.ClassName == "Circle" then
        local dx, dy = mx - obj.X, my - obj.Y
        return (dx*dx + dy*dy) <= obj.Radius * obj.Radius
    end
    return false
end

-- ========== 交互逻辑 ==========

local UserInput = ""
local IsTyping = false
local Dragging = false
local DragOffset = Vector2.new(0, 0)
local UIHidden = false

-- 构建 UI
BuildUI()

-- 鼠标点击
UIS.InputBegan:Connect(function(input, processed)
    if processed or UIHidden then return end
    if input.UserInputType == Enum.UserInputType.MouseButton1 then
        local mx, my = UIS:GetMouseLocation().X, UIS:GetMouseLocation().Y
        
        -- 关闭按钮
        if IsOver(UI.Close) then
            UIHidden = true
            for _, obj in pairs(UI) do
                if type(obj) == "table" and obj.Visible ~= nil then
                    obj.Visible = false
                end
            end
            return
        end
        
        -- 发送按钮
        if IsOver(UI.Send) and UserInput ~= "" then
            UI.Status.Text = "🤔 正在向 AI 请求..."
            UI.CodeText.Text = "生成中..."
            
            AskAI(UserInput, function(code, err)
                if code then
                    UI.CodeText.Text = code
                    UI.Status.Text = "✅ 生成成功！点击「执行」运行"
                else
                    UI.CodeText.Text = "-- 错误: " .. (err or "未知")
                    UI.Status.Text = "❌ " .. (err or "失败")
                end
            end)
            return
        end
        
        -- 执行按钮
        if IsOver(UI.Exec) and UI.CodeText.Text ~= "等待 AI 生成代码..." and UI.CodeText.Text ~= "生成中..." then
            UI.Status.Text = "⚡ 正在执行..."
            local ok, result = RunCode(UI.CodeText.Text)
            if ok then
                UI.Status.Text = "✅ 执行成功！"
            else
                UI.Status.Text = "❌ 执行失败"
                UI.CodeText.Text = UI.CodeText.Text .. "\n\n-- " .. tostring(result)
            end
            return
        end
        
        -- 清空按钮
        if IsOver(UI.Clear) then
            UI.CodeText.Text = "等待 AI 生成代码..."
            UI.Status.Text = "已清空"
            return
        end
        
        -- 输入框点击
        if IsOver(UI.InputBg) then
            IsTyping = true
            UI.Placeholder.Visible = false
            UI.Status.Text = "⌨️ 输入中... Enter 发送"
            return
        else
            IsTyping = false
            if UserInput == "" then
                UI.Placeholder.Visible = true
            end
        end
        
        -- 标题栏拖动
        if IsOver(UI.TitleBg) then
            Dragging = true
            DragOffset = Vector2.new(mx - Window.X, my - Window.Y)
        end
    end
end)

-- 键盘输入
UIS.InputBegan:Connect(function(input, processed)
    if processed or not IsTyping or UIHidden then return end
    if input.UserInputType == Enum.UserInputType.Keyboard then
        local key = input.KeyCode
        
        if key == Enum.KeyCode.Backspace then
            UserInput = UserInput:sub(1, -2)
        elseif key == Enum.KeyCode.Return then
            IsTyping = false
            if UserInput ~= "" then
                UI.Status.Text = "🤔 正在向 AI 请求..."
                AskAI(UserInput, function(code, err)
                    if code then
                        UI.CodeText.Text = code
                        UI.Status.Text = "✅ 生成成功！"
                    else
                        UI.CodeText.Text = "-- 错误: " .. (err or "未知")
                        UI.Status.Text = "❌ " .. (err or "失败")
                    end
                end)
            end
        elseif key == Enum.KeyCode.Escape then
            IsTyping = false
            if UserInput == "" then
                UI.Placeholder.Visible = true
            end
        else
            local char = input.KeyCode.Name
            if #char == 1 then
                UserInput = UserInput .. char
            elseif char == "Space" then
                UserInput = UserInput .. " "
            end
        end
        UI.InputText.Text = UserInput
    end
end)

-- 鼠标移动（拖动）
UIS.InputChanged:Connect(function(input, processed)
    if processed or UIHidden then return end
    if input.UserInputType == Enum.UserInputType.MouseMovement and Dragging then
        local mx, my = UIS:GetMouseLocation().X, UIS:GetMouseLocation().Y
        Window.X = mx - DragOffset.X
        Window.Y = my - DragOffset.Y
        UpdatePositions()
    end
end)

-- 鼠标释放
UIS.InputEnded:Connect(function(input, processed)
    if input.UserInputType == Enum.UserInputType.MouseButton1 then
        Dragging = false
    end
end)

-- 渲染循环
RunService.RenderStepped:Connect(function()
    if UIHidden then return end
    
    -- 光标闪烁
    if IsTyping then
        local cursor = (tick() % 1 < 0.5) and "|" or " "
        UI.InputText.Text = UserInput .. cursor
    end
end)
